cmake_minimum_required(VERSION 3.22)
project(LanternPairing LANGUAGES C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS pairing_compile=4)
set(CMAKE_JOB_POOL_COMPILE pairing_compile)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPENSSL_NO_ASM ON CACHE BOOL "" FORCE)
if(NOT BORINGSSL_SOURCE)
  message(FATAL_ERROR "Run scripts/build-pairing.ps1 to restore the pinned BoringSSL source")
endif()
add_subdirectory("${BORINGSSL_SOURCE}" boringssl EXCLUDE_FROM_ALL)
add_library(lantern_pake SHARED pairing.cc)
target_include_directories(lantern_pake PRIVATE "${BORINGSSL_SOURCE}" "${BORINGSSL_SOURCE}/include")
target_link_libraries(lantern_pake PRIVATE crypto)
if(NOT ANDROID)
  add_executable(lantern_pairing_test EXCLUDE_FROM_ALL rfc9383_test.cc)
  target_include_directories(lantern_pairing_test PRIVATE "${BORINGSSL_SOURCE}" "${BORINGSSL_SOURCE}/include")
  target_link_libraries(lantern_pairing_test PRIVATE crypto)
endif()
if(ANDROID)
  target_compile_options(lantern_pake PRIVATE -fvisibility=hidden)
else()
  find_package(JNI REQUIRED)
  target_include_directories(lantern_pake PRIVATE ${JNI_INCLUDE_DIRS})
endif()
